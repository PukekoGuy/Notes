\begin{theorem}
    Let $X$ be a scheme, let
    $\cU = \{U_\lambda\}_{\lambda \in \Lambda}$ be an open cover
    of $X$, and fix a rank $n$. Then there is a bijection
    \begin{equation}
        \left\{
            \begin{gathered}
                \textnormal{locally free $\cO_X$-modules of rank $n$}\\
                \textnormal{trivialised over $\cU$}
            \end{gathered}
        \right\}/\cong
        \quad\longleftrightarrow\quad
        \check H^1(\cU,\GL_n(\cO_X)).
    \end{equation}
\end{theorem}

\begin{proof}
    Write $U_{\lambda\mu}=U_\lambda\cap U_\mu$ and
    $U_{\lambda\mu\nu}=U_\lambda\cap U_\mu\cap U_\nu$.
    We use $\cO_U=\cO_X|_U$ for each open subset $U\subseteq X$. To specify the distinction between matrices and sheaf morphisms, if $A\in\GL_n(\cO_X(U))$, let $\mathrm m_A:\cO_U^{\oplus n}\longrightarrow \cO_U^{\oplus n}$ denote the automorphism of $\cO_U$-modules whose component on an open set $V\subseteq U$ is $(\mathrm m_A)_V(s)=(A|_V)s$. Note that $\Aut_{\cO_U}(\cO_U^{\oplus n})
        \cong \GL_n(\cO_X(U))
        =\GL_n(\cO_X)(U)$, where $\Aut_{\cO_U}$ denotes automorphisms of sheaves of
    $\cO_U$-modules. \\

    We first construct the forward map $\alpha$.
    Let $\sF$ be a locally free $\cO_X$-module of rank $n$ trivialised over $\cU$. Choose isomorphisms of
    $\cO_{U_\lambda}$-modules
    \[
        \phi_\lambda:
        \cO_{U_\lambda}^{\oplus n}
        \longrightarrow \sF|_{U_\lambda}.
    \]
    For every open set $V\subseteq U_\lambda$, the corresponding
    map on sections is an isomorphism of $\cO_X(V)$-modules
    \[
        (\phi_\lambda)_V:
        \cO_X(V)^{\oplus n}\longrightarrow \sF(V).
    \]

    Define
    \[
        T_{\lambda\mu}
        \in \GL_n(\cO_X(U_{\lambda\mu}))
    \]
    by the identity of sheaf morphisms
    \[
        \mathrm m_{T_{\lambda\mu}}
        =
        (\phi_\lambda|_{U_{\lambda\mu}})^{-1}
        \circ
        (\phi_\mu|_{U_{\lambda\mu}}).
    \]

    On $U_{\lambda\mu\nu}$, we have
    \[
        \mathrm m_{T_{\lambda\mu}T_{\mu\nu}}
        =
        (\phi_\lambda^{-1}\circ\phi_\mu)
        \circ
        (\phi_\mu^{-1}\circ\phi_\nu)
        =
        \phi_\lambda^{-1}\circ\phi_\nu
        =
        \mathrm m_{T_{\lambda\nu}}.
    \]
    Hence
    \[
        T_{\lambda\mu}T_{\mu\nu}=T_{\lambda\nu}.
    \]
    We also have $T_{\lambda\lambda}=I_n$ and
    $T_{\mu\lambda}=T_{\lambda\mu}^{-1}$. Therefore
    \[
        (T_{\lambda\mu})_{\lambda,\mu\in\Lambda}
        \in\check Z^1(\cU,\GL_n(\cO_X)).
    \]

    Now suppose that $\sF'$ is another locally free
    $\cO_X$-module of rank $n$ trivialised over $\cU$,
    with chosen trivialisations $\phi_\lambda'$ and
    transition matrices $T_{\lambda\mu}'$, and let
    \[
        \psi:\sF\longrightarrow\sF'
    \]
    be an isomorphism of $\cO_X$-modules.
    Write $\psi_\lambda=\psi|_{U_\lambda}$ and define $h_\lambda\in\GL_n(\cO_X(U_\lambda))$ by
    \[
        \mathrm m_{h_\lambda}
        =
        (\phi_\lambda')^{-1}
        \circ\psi_\lambda\circ\phi_\lambda.
    \]
    Thus
    \[
        (h_\lambda)_{\lambda\in\Lambda}
        \in
        \check C^0(\cU,\GL_n(\cO_X))
        =
        \prod_{\lambda\in\Lambda}
        \GL_n(\cO_X(U_\lambda)).
    \]
    On $U_{\lambda\mu}$, the restrictions of $\psi_\lambda$
    and $\psi_\mu$ agree, so
    \begin{align*}
        \mathrm m_{h_\lambda T_{\lambda\mu}}
        &=
        (\phi_\lambda')^{-1}
        \circ\psi_\lambda\circ\phi_\mu\\
        &=
        (\phi_\lambda')^{-1}
        \circ\psi_\mu\circ\phi_\mu\\
        &=
        \mathrm m_{T_{\lambda\mu}'h_\mu}.
    \end{align*}
    Consequently, $h_\lambda T_{\lambda\mu}=T_{\lambda\mu}'h_\mu$, which is precisely the equivalence relation defining
    $\check H^1(\cU,\GL_n(\cO_X))$. We therefore obtain a well-defined map
    \[
        \alpha:[\sF]\longmapsto[(T_{\lambda\mu})].
    \]

    To prove injectivity, suppose that the cocycles associated
    to $\sF$ and $\sF'$ represent the same class. Thus there
    exists a $0$-cochain $(h_\lambda)_{\lambda\in\Lambda}$ such that
    \[
        h_\lambda T_{\lambda\mu}
        =
        T_{\lambda\mu}'h_\mu
    \]
    on every overlap. Define isomorphisms of
    $\cO_{U_\lambda}$-modules
    \[
        \psi_\lambda
        =
        \phi_\lambda'\circ\mathrm m_{h_\lambda}
        \circ\phi_\lambda^{-1}:
        \sF|_{U_\lambda}\longrightarrow\sF'|_{U_\lambda}.
    \]
    On $U_{\lambda\mu}$, we have
    \begin{align*}
        \psi_\lambda
        &=
        \phi_\lambda'\circ\mathrm m_{h_\lambda}
        \circ\mathrm m_{T_{\lambda\mu}}
        \circ\phi_\mu^{-1}\\
        &=
        \phi_\lambda'\circ\mathrm m_{T_{\lambda\mu}'}
        \circ\mathrm m_{h_\mu}\circ\phi_\mu^{-1}\\
        &=
        \phi_\mu'\circ\mathrm m_{h_\mu}\circ\phi_\mu^{-1}\\
        &=
        \psi_\mu.
    \end{align*}
    Hence the $\psi_\lambda$ glue to an isomorphism
    $\psi:\sF\to\sF'$ of $\cO_X$-modules; their inverses
    also agree on overlaps and glue to its inverse.
    Thus $\alpha$ is injective.\\

    For the reverse construction, let
    \[
        (T_{\lambda\mu})_{\lambda,\mu\in\Lambda}
        \in\check Z^1(\cU,\GL_n(\cO_X)).
    \]
    For each open set $V\subseteq X$, define
    \begin{equation}
        \sF(V)=
        \left\{
            (s_\lambda)_{\lambda\in\Lambda}
            \in
            \prod_{\lambda\in\Lambda}
            \cO_X(V\cap U_\lambda)^{\oplus n}
            \ \middle|\
            s_\lambda=T_{\lambda\mu}s_\mu
            \text{ on }V\cap U_{\lambda\mu}
        \right\}.
    \end{equation}
    For open sets $W\subseteq V$, define restriction maps
    \[
        \sF(V)\longrightarrow\sF(W),
        \qquad
        (s_\lambda)_\lambda
        \longmapsto
        (s_\lambda|_{W\cap U_\lambda})_\lambda.
    \]
    These maps preserve the compatibility condition and satisfy the presheaf axioms. Addition is defined componentwise, and for
    $f\in\cO_X(V)$ scalar multiplication is defined by
    \[
        f(s_\lambda)_\lambda
        =
        \bigl((f|_{V\cap U_\lambda})s_\lambda\bigr)_\lambda.
    \]
    These operations preserve the compatibility condition because multiplication by each $T_{\lambda\mu}$ is linear.
    Thus $\sF$ is a presheaf of $\cO_X$-modules. The separation and gluing axioms follow componentwise from those for $\cO_X^{\oplus n}$.\\

    We now construct isomorphisms of sheaves
    \[
        \phi_\lambda:
        \cO_{U_\lambda}^{\oplus n}
        \longrightarrow\sF|_{U_\lambda}.
    \]
    For every open set $V\subseteq U_\lambda$, define
    \[
        (\phi_\lambda)_V:
        \cO_X(V)^{\oplus n}\longrightarrow\sF(V),
        \qquad
        s\longmapsto
        \left(
            \left.T_{\mu\lambda}\right|_{V\cap U_\mu}
            \left.s\right|_{V\cap U_\mu}
        \right)_{\mu\in\Lambda}.
    \]
    This family satisfies the compatibility condition because
    \[
        T_{\mu\nu}T_{\nu\lambda}=T_{\mu\lambda}.
    \]
    The map $(\phi_\lambda)_V$ is an isomorphism of
    $\cO_X(V)$-modules, with inverse
    \[
        (s_\mu)_{\mu\in\Lambda}\longmapsto s_\lambda.
    \]
    Indeed, $T_{\lambda\lambda}=I_n$, and every compatible family satisfies $s_\mu=T_{\mu\lambda}s_\lambda$. These maps commute with restriction, so $\phi_\lambda$
    is an isomorphism of sheaves of $\cO_{U_\lambda}$-modules.
    Therefore $\sF$ is locally free of rank $n$ and is
    trivialised over $\cU$.

    On $U_{\lambda\mu}$, the $\lambda$-component of
    $\phi_\mu(s)$ is $T_{\lambda\mu}s$. Consequently,
    \[
        \phi_\lambda^{-1}\circ\phi_\mu
        =
        \mathrm m_{T_{\lambda\mu}},
    \]
    so the transition matrices of the constructed sheaf
    are exactly the original $T_{\lambda\mu}$.

    Finally, if
    \[
        T_{\lambda\mu}'
        =
        h_\lambda T_{\lambda\mu}h_\mu^{-1},
    \]
    the componentwise maps
    \[
        \sF(V)\longrightarrow\sF'(V),
        \qquad
        (s_\lambda)_\lambda
        \longmapsto
        \bigl((h_\lambda|_{V\cap U_\lambda})s_\lambda\bigr)_\lambda
    \]
    define an isomorphism of $\cO_X$-modules, with inverse
    given componentwise by $h_\lambda^{-1}$.
    Thus the reverse construction defines a well-defined map
    \[
        \beta:[(T_{\lambda\mu})]\longmapsto[\sF].
    \]
    Since $\alpha\circ\beta=\Id$, the map $\alpha$ is
    surjective. Together with injectivity, this proves the
    claimed bijection.
\end{proof}

\begin{theorem}
    If $X$ is a topological space then there is a bijection
    \begin{equation}
        \left\{
            \begin{gathered}
            \textnormal{real vector }\\
            \textnormal{bundles on $X$}
            \end{gathered}
        \right\}/\cong \quad\longleftrightarrow\quad \left\{
            \begin{gathered}
            \textnormal{locally free sheaves} \\
            \textnormal{of }\cC_{X,\R}\textnormal{-modules } \textnormal{on $X$}
            \end{gathered}
        \right\}/\cong
    \end{equation}
\end{theorem}
\begin{proof}

We first construct the $\rightarrow$ map $\alpha$. Let $E \overset{\pi}\to X$ be a real vector bundle of rank $n$, trivialised over an open cover
$\cU = \{U_\lambda\}_{\lambda \in \Lambda}$ by homeomorphisms
\[
    h_\lambda:\pi^{-1}(U_\lambda) \longrightarrow U_\lambda \times \R^n.
\]
For each $x \in X$, write $E_x = \pi^{-1}(x)$. The vector space structures on the fibres define maps
\begin{align}
    a:E \times_X E \longrightarrow E,
    \qquad (e,e') \longmapsto e+e',\\
    m:\R \times E \longrightarrow E,
    \qquad (r,e) \longmapsto re,
\end{align}
where
\[
    E \times_X E
    = \{(e,e') \in E \times E \mid \pi(e)=\pi(e')\}
\]
has the subspace topology. These maps are globally well defined:
addition is performed in the common fibre of $e$ and $e'$, and scalar
multiplication is performed in the fibre containing $e$. In particular,
their definitions do not depend on any choice of trivialisation.

We claim that both maps are continuous. For each $\lambda$, the trivialisation $h_\lambda$ induces a homeomorphism
\[
    \pi^{-1}(U_\lambda) \times_{U_\lambda} \pi^{-1}(U_\lambda)
    \longrightarrow U_\lambda \times \R^n \times \R^n,
\]
sending $(e,e')$ to $(x,v,w)$ when
$h_\lambda(e)=(x,v)$ and $h_\lambda(e')=(x,w)$.
Since $h_\lambda$ is linear on each fibre, the restriction of $a$
is represented in these coordinates by
\[
    (x,v,w) \longmapsto (x,v+w).
\]
Likewise, using the homeomorphism
\[
    \R \times \pi^{-1}(U_\lambda)
    \longrightarrow U_\lambda \times \R \times \R^n,
    \qquad
    (r,e) \longmapsto (\pi(e),r,\pr_2(h_\lambda(e))),
\]
the restriction of $m$ is represented by
\[
    (x,r,v) \longmapsto (x,rv).
\]
These coordinate maps are continuous. Moreover, the sets
$\pi^{-1}(U_\lambda) \times_{U_\lambda} \pi^{-1}(U_\lambda)$
form an open cover of $E \times_X E$, and the sets
$\R \times \pi^{-1}(U_\lambda)$ form an open cover of $\R \times E$.
Thus $a$ and $m$ are continuous on their entire domains.

For each open set $V \subseteq X$, define
\[
    \cE(V)
    =
    \{s:V \to E \text{ continuous} \mid \pi \circ s = i_{V,X}\},
\]
with restriction maps given by function restriction. Define addition
and multiplication by $f \in \cC_{X,\R}(V)$ pointwise:
\[
    (s+t)(x)=s(x)+t(x),
    \qquad
    (fs)(x)=f(x)s(x).
\]
These operations produce sections over $V$, since their values lie in $E_x$, which implies that $\pi \circ s=i_{V,X}$ can be verified locally. They also preserve continuity: the map
\[
    V \longrightarrow E \times_X E,
    \qquad x \longmapsto (s(x),t(x)),
\]
is continuous, and its composition with $a$ is $s+t$; similarly,
$fs$ is the composition of the continuous map
\[
    V \longrightarrow \R \times E,
    \qquad x \longmapsto (f(x),s(x)),
\]
with $m$.

The zero section $x \mapsto 0_{E_x}$ is continuous, since in every trivialisation it is given by $x \mapsto (x,0)$, and the additive inverse of $s$ is the continuous section $(-1)s$. All module axioms
hold pointwise by the vector space axioms in each fibre. Hence $\cE(V)$ is a $\cC_{X,\R}(V)$-module. These operations commute with restriction, so $\cE$ is a presheaf of $\cC_{X,\R}$-modules. Continuous sections that agree on overlaps glue uniquely to a
continuous section, so $\cE$ is a sheaf of
$\cC_{X,\R}$-modules.

For each $\lambda$, the trivialisation $h_\lambda$ induces an isomorphism of sheaves of $\cC_{X,\R}|_{U_\lambda}$-modules
\[
    \phi_\lambda:
    \cC_{X,\R}^{\oplus n}|_{U_\lambda} \longrightarrow \cE|_{U_\lambda}.
\]

For every open set $V \subseteq U_\lambda$, define $(\phi_\lambda)_V$ by sending a continuous map $f:V \to \R^n$ to the continuous section
\[
    x \longmapsto h_\lambda^{-1}(x,f(x)).
\]
It has an inverse, given by
\[
    (\phi_\lambda)_V^{-1}(s)
    =
    \bigl(x \longmapsto \pr_2(h_\lambda(s(x)))\bigr).
\]
Both constructions commute with restrictions, so $\phi_\lambda$ is an isomorphism, and $\cE$ is locally free of rank $n$. \\

Finally, if $F:E \to \widetilde E$ is a vector bundle isomorphism
over $X$, composition with $F$ induces an isomorphism of sheaves
of $\cC_{X,\R}$-modules
\[
    \cE \longrightarrow \widetilde{\cE},
    \qquad s \longmapsto F \circ s.
\]
Consequently, the assignment $\alpha([E])=[\cE]$ is well defined on isomorphism classes.\\

We now consider the $\leftarrow$ direction, $\beta$. Let $\cE$ be a locally free sheaf of $\cC_{X,\R}$-modules, trivialised over an open cover $\cU=\{U_\lambda\}_{\lambda \in \Lambda}$ by isomorphisms of sheaves of $\cC_{X,\R}$-modules
\begin{equation}
    \phi_\lambda:
    \cC_{X,\R}^{\oplus n}|_{U_\lambda} \longrightarrow \cE|_{U_\lambda}.
\end{equation}
On $U_{\lambda \mu}$, we have a matrix 
\begin{equation}
    T_{\lambda \mu} \in \GL_n(\C_{X,\R}(U_{\lambda\mu})),
\end{equation}
corresponding to the isomorphism of sheaves $\mathrm m_{T_\lambda \mu} = \phi_\lambda^{-1} \phi_\mu$. Composition of these morphisms gives
\begin{equation}\label{thm2:eq:transition functions}
    T_{\lambda\lambda}=I_n,
    \qquad
    T_{\mu\lambda}=T_{\lambda\mu}^{-1},
    \qquad
    T_{\lambda\mu}T_{\mu\nu}=T_{\lambda\nu},
\end{equation}

Let us define the total space
\[
    E=
    \left(
        \coprod_{\lambda\in\Lambda}U_\lambda\times\R^n
    \right)\Big/\sim,
\]
where
\[
    (x,v)_\lambda\sim(y,w)_\mu
    \quad\Longleftrightarrow\quad
    x=y
    \quad\text{and}\quad
    v=T_{\lambda\mu}(x)w.
\]
By Equation \ref{thm2:eq:transition functions}, this relation is reflexive, symmetric, and transitive. Let us denote the quotient map by
\[
    q:\coprod_{\lambda\in\Lambda}U_\lambda\times\R^n
    \longrightarrow E
\]
and define
\[
    \pi:E\longrightarrow X,
    \qquad
    [(x,v)_\lambda]\longmapsto x.
\]

Gluing along the transition homeomorphisms gives local
trivialisations
\[
    h_\lambda:\pi^{-1}(U_\lambda)
    \longrightarrow U_\lambda\times\R^n,
    \qquad
    [(x,v)_\mu]\longmapsto(x,T_{\lambda\mu}(x)v),
\]
with inverses
\[
    (x,v)\longmapsto[(x,v)_\lambda].
\]
The vector space structures on $\R^n$ induce vector space
structures on the fibres of $\pi$, independently of the
chosen chart, because every $T_{\lambda\mu}(x)$ is linear.
Thus $E\overset{\pi}\to X$ is a real vector bundle of rank $n$.

Let $\widetilde{\cE}$ be its sheaf of continuous sections.
For an open set $V\subseteq X$ and $s\in\cE(V)$, write
\[
    \overline s_\lambda
    =
    (\phi_\lambda^{-1})_{V\cap U_\lambda}
    (s|_{V\cap U_\lambda}).
\]
These coordinate sections satisfy
\[
    \overline s_\lambda
    =
    T_{\lambda\mu}\overline s_\mu
\]
on overlaps, so the formula
\[
    \eta_V(s)(x)
    =
    [(x,\overline s_\lambda(x))_\lambda],
    \qquad x\in V\cap U_\lambda,
\]
defines a continuous section of $E$ over $V$.

Conversely, a continuous section $t$ of $E$ over $V$ has
coordinate maps
\[
    \overline t_\lambda(x)=\pr_2(h_\lambda(t(x))),
    \qquad x\in V\cap U_\lambda,
\]
satisfying the same compatibility condition. Hence
\[
    (\phi_\lambda)_{V\cap U_\lambda}(\overline t_\lambda)
    \in\cE(V\cap U_\lambda)
\]
agree on overlaps and glue uniquely to an element of $\cE(V)$.
This construction is inverse to $\eta_V$. These maps are
$\cC_{X,\R}(V)$-linear and commute with restriction, giving
an isomorphism
\[
    \eta:\cE\xrightarrow{\sim}\widetilde{\cE}.
\]

To check independence of choices, suppose that
$\theta:\cE\to\cE'$ is an isomorphism, and choose local
trivialisations $\phi_\lambda$ and $\phi_\lambda'$ on a
common refinement of their covers. Let $H_\lambda$ be
the matrix representing the sheaf isomorphism
\[
    (\phi_\lambda')^{-1}
    \circ\theta|_{U_\lambda}\circ\phi_\lambda.
\]
Then
\[
    H_\lambda T_{\lambda\mu}
    =
    T_{\lambda\mu}'H_\mu,
\]
so the local maps
\[
    (x,v)\longmapsto(x,H_\lambda(x)v)
\]
glue to a vector bundle isomorphism between the reconstructed
bundles. Taking $\theta=\Id_{\cE}$ also proves independence
of the chosen trivialisations; passing to a refinement
does not change the glued bundle up to isomorphism.
Therefore
\[
    \beta:[\cE]\longmapsto[E]
\]
is well defined, and the isomorphism $\eta$ proves that
$\alpha\circ\beta=\Id$.

Finally, if we start with a vector bundle $E$ with
trivialisations $h_\lambda$, the induced trivialisations
of its section sheaf give the same transition matrices.
The reconstructed bundle is therefore isomorphic to $E$
via
\[
    [(x,v)_\lambda]\longmapsto h_\lambda^{-1}(x,v).
\]
Thus $\beta\circ\alpha=\Id$ on isomorphism classes,
completing the proof.

\end{proof}
